TFRKPHN_dept_start = {
	if = {
		limit = { NOT = { has_country_flag = TFRKPHN_dept_started } }
		set_country_flag = TFRKPHN_dept_started
		set_variable = { TFRKPHN_dept_t_pm = 50 }
		set_variable = { TFRKPHN_dept_t_fm = 50 }

		set_variable = { TFRKPHN_dept_t_im = 50 }
		set_variable = { TFRKPHN_dept_t_nis = 50 }
		set_variable = { TFRKPHN_dept_t_dm = 50 }

	}
