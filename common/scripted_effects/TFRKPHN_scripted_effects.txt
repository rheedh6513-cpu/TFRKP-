# [안내] 여러 곳에서 같이 쓰는 효과 묶음 (트리 교체, 한일 핵 패널 진행 등). 전역 변수는 set_variable = { global.이름 = 값 }.
# [안내] 수정 방법: 모드 폴더의 TUTORIAL_KO.md 7~9장 참고. 새 이름에는 TFRKPHN_ 접두어를 붙이세요.
TFRKPHN_hourglass_init = {
	if = {
		limit = { NOT = { has_country_flag = TFRKPHN_hourglass_init } }
		set_country_flag = TFRKPHN_hourglass_init
		set_variable = { TFRKPHN_hourglass_step = 0 }
		set_variable = { TFRKPHN_hourglass_frame = 1 }
	}
}

TFRKPHN_hourglass_advance = {
	TFRKPHN_hourglass_init = yes
	add_to_variable = { TFRKPHN_hourglass_step = 1 }
	set_variable = { TFRKPHN_hourglass_frame = TFRKPHN_hourglass_step }
	add_to_variable = { TFRKPHN_hourglass_frame = 1 }
	# 결정 11개를 다 눌렀으면 마지막 결정이 끝나는 14일 뒤에 마무리 이벤트 (hourglass_end_build.py)
	if = {
		limit = { check_variable = { TFRKPHN_hourglass_step > 10.5 } NOT = { has_country_flag = TFRKPHN_hourglass_end_sent } }
		set_country_flag = TFRKPHN_hourglass_end_sent
		country_event = { id = tfrkphn.528 days = 14 }
	}
}

TFRKPHN_hong_start_effect = {
	if = {
		limit = { NOT = { has_character = KOR_hong_joon_pyo_char } }
		recruit_character = KOR_hong_joon_pyo_char
	}
	if = {

		KOR_hong_joon_pyo_char = { promote_character = judicial_conservatism }
	}
	set_country_flag = TFRKP_HN_hong_president
	# 당선 뒤 2주마다 차례로 뜨는 사건 (예전에는 중점 보상이었다)
	if = {
		limit = { NOT = { has_country_flag = TFRKPHN_hong_auto_events_sent } }
		set_country_flag = TFRKPHN_hong_auto_events_sent
		country_event = { id = tfrkphn.21 days = 14 }
		country_event = { id = tfrkphn.22 days = 28 }
		country_event = { id = tfrkphn.23 days = 42 }
		country_event = { id = tfrkphn.24 days = 56 }
		country_event = { id = tfrkphn.313 days = 70 }
		country_event = { id = tfrkphn.314 days = 105 }
		country_event = { id = tfrkphn.315 days = 140 }

	# 제20대 대선에서 이낙연을 이긴 경우: 3일 뒤 정의당 당사 사건 '늘 그래 왔으니까'

		limit = {
			has_country_flag = TFRKP_HN_DPK_lee_nakyeon
			NOT = { has_global_flag = TFRKPHN_justice_hq_fired }
		}
		set_global_flag = TFRKPHN_justice_hq_fired
		country_event = { id = tfrkphn.307 days = 3 }

		limit = { NOT = { has_focus_tree = TFRKPHN_hong_tree } }
		load_focus_tree = { tree = TFRKPHN_hong_tree keep_completed = no }
	}
	mark_focus_tree_layout_dirty = yes
	# 지역별 지지도 시작값: 전국 평균 약 62% (취임 직후). 막말 구설수와 무작위 하락으로 40%대까지 떨어지고, 승전하면 오른다
	set_variable = { TFRKPHN_support_seoul = 64 }
	set_variable = { TFRKPHN_support_gyeonggi = 60 }
	set_variable = { TFRKPHN_support_gangwon = 68 }
	set_variable = { TFRKPHN_support_chungcheong = 65 }
	set_variable = { TFRKPHN_support_jeonbuk = 28 }

	set_variable = { TFRKPHN_support_jeju = 56 }
	set_variable = { TFRKPHN_select_region_var = 0 }
	TFRKPHN_hong_update_support_display = yes
	# 2주마다 무작위 지역 지지도 하락 시작
	country_event = { id = tfrkphn.100 days = 14 }
}

# ===== 대통령 임기 (개헌 4년 중임제 / 기존 5년 단임제) =====
# 헌법 제128조 제2항: 중임변경 개헌은 제안 당시 대통령에게는 효력이 없다.
# 그래서 홍준표 정부에서 개헌(TFRKPHN_const_term)이 되면, 2027년에 뽑힌 다음 정권부터 4년 중임제가 적용된다.
TFRKPHN_term_start_effect = {
	set_country_flag = TFRKPHN_pres_cycle
	set_country_flag = TFRKPHN_hong_term_over	# 홍준표 임기 끝: 홍준표 트리 안전장치(tfrkphn.12)를 끈다
	# 홍준표 정부의 지지도 하락·개헌 미션은 새 정권에서 끝낸다
	hidden_effect = {
		if = { limit = { has_active_mission = TFRKPHN_support_decay_mission } remove_mission = TFRKPHN_support_decay_mission }
		if = { limit = { has_active_mission = TFRKPHN_amend_opposition_moves } remove_mission = TFRKPHN_amend_opposition_moves }
		if = { limit = { has_active_mission = TFRKPHN_amend_deadline } remove_mission = TFRKPHN_amend_deadline }
		clr_country_flag = TFRKPHN_amend_active
		clr_country_flag = TFRKPHN_amend_ongoing
	}
	clr_country_flag = TFRKPHN_pres_term2
	if = {
		limit = { has_country_flag = TFRKPHN_const_term }
		set_country_flag = TFRKPHN_term4_in_force
	}
	if = {
		limit = { has_country_flag = TFRKPHN_term4_in_force }
		set_variable = { TFRKPHN_term_days = 1461 }	# 대선일부터 4년 뒤 다음 대선
		custom_effect_tooltip = TFRKPHN_term4_tt
	}
	else = {
		set_variable = { TFRKPHN_term_days = 1826 }	# 대선일부터 5년 뒤 다음 대선
		custom_effect_tooltip = TFRKPHN_term5_tt
	}
}

# 북한 내전 세력끼리 전쟁 다시 잇기 (괴뢰화하면 전쟁이 끊기므로)
TFRKPHN_nk_rewar_effect = {
		# 이북5도 vs 북한(김여정), 이북5도 vs 군사위원회, 군사위원회 vs 북한
		random_country = {
			limit = { has_country_flag = TFRKPHN_free_joseon }
			save_event_target_as = TFRKPHN_ibuk5do_t
		}
		random_country = {
			limit = { has_country_flag = TFRKPHN_kpa_junta exists = yes }
			save_event_target_as = TFRKPHN_kpa_t
		}
		if = {
			limit = { has_event_target = TFRKPHN_ibuk5do_t country_exists = PRK }
			event_target:TFRKPHN_ibuk5do_t = {
				if = { limit = { NOT = { has_war_with = PRK } } declare_war_on = { target = PRK type = annex_everything } }
			}
		}
		if = {
			limit = { has_event_target = TFRKPHN_ibuk5do_t has_event_target = TFRKPHN_kpa_t }
			event_target:TFRKPHN_ibuk5do_t = {
				if = { limit = { NOT = { has_war_with = event_target:TFRKPHN_kpa_t } } declare_war_on = { target = event_target:TFRKPHN_kpa_t type = annex_everything } }
			}
		}
		if = {
			limit = { has_event_target = TFRKPHN_kpa_t country_exists = PRK }
			event_target:TFRKPHN_kpa_t = {
				if = { limit = { NOT = { has_war_with = PRK } } declare_war_on = { target = PRK type = annex_everything } }
			}
		}
}
