TFRKPHN_sim_seats_calc = {
	if = { limit = { NOT = { has_variable = KOR_NatAsm_JP_seats } } set_variable = { KOR_NatAsm_JP_seats = 0 } }
	if = { limit = { NOT = { has_variable = KOR_NatAsm_DP_seats } } set_variable = { KOR_NatAsm_DP_seats = 0 } }
	set_temp_variable = { TFRKPHN_sim_t = KOR_NatAsm_DP_seats }
	multiply_temp_variable = { TFRKPHN_sim_t = TFRKPHN_sim_dem }
	divide_temp_variable = { TFRKPHN_sim_t = 100 }
	set_variable = { TFRKPHN_sim_seats = KOR_NatAsm_JP_seats }
	add_to_variable = { TFRKPHN_sim_seats = TFRKPHN_sim_t }
	round_variable = TFRKPHN_sim_seats
	set_variable = { TFRKPHN_sim_seats_f = TFRKPHN_sim_seats }
	divide_variable = { TFRKPHN_sim_seats_f = 10 }
	subtract_from_variable = { TFRKPHN_sim_seats_f = 0.49 }
	round_variable = TFRKPHN_sim_seats_f
	add_to_variable = { TFRKPHN_sim_seats_f = 1 }
	clamp_variable = { var = TFRKPHN_sim_seats_f min = 1 max = 31 }
}
