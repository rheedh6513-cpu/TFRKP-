# [안내] 토요일 광장. 광장 값을 0~100으로 맞추고 국민정신 값을 고친다. cho_build.py(cho_rally_part.py)가 만든다.
TFRKPHN_cr_refresh = {
	clamp_variable = { var = TFRKPHN_cr min = 0 max = 100 }
	set_variable = { TFRKPHN_cr_pp = TFRKPHN_cr }
	subtract_from_variable = { TFRKPHN_cr_pp = 50 }
	set_variable = { TFRKPHN_cr_ws = TFRKPHN_cr_pp }
	multiply_variable = { TFRKPHN_cr_pp = 0.003 }
	multiply_variable = { TFRKPHN_cr_ws = 0.0005 }
	if = { limit = { NOT = { has_dynamic_modifier = { modifier = TFRKPHN_cr_modifier } } } add_dynamic_modifier = { modifier = TFRKPHN_cr_modifier } }
	force_update_dynamic_modifier = yes
}
# 중점이 광장 값을 움직일 때 쓴다 (탭이 열려 있을 때만)
TFRKPHN_cr_add = {
	if = { limit = { has_country_flag = TFRKPHN_cr_init NOT = { has_country_flag = TFRKPHN_cr_closed } } add_to_variable = { TFRKPHN_cr = cr_add } TFRKPHN_cr_refresh = yes }
}
