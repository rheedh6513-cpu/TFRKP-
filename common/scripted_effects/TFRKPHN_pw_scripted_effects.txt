# [안내] 승전 뒤 중점을 끝낼 때 국민정신 「승전의 유산」에 수치를 더한다. pw_build.py가 만든다.
TFRKPHN_pw_grant_prosperity = {
	custom_effect_tooltip = TFRKPHN_pw_grant_prosperity_tt
	hidden_effect = {
		add_to_variable = { TFRKPHN_pw_industrial_capacity_factory = 0.1 }
		add_to_variable = { TFRKPHN_pw_stability_factor = 0.03 }
		if = { limit = { NOT = { has_dynamic_modifier = { modifier = TFRKPHN_pw_legacy } } } add_dynamic_modifier = { modifier = TFRKPHN_pw_legacy } }
		force_update_dynamic_modifier = yes
	}
}
TFRKPHN_pw_grant_defense = {
	custom_effect_tooltip = TFRKPHN_pw_grant_defense_tt
	hidden_effect = {
		add_to_variable = { TFRKPHN_pw_army_attack_factor = 0.05 }
		add_to_variable = { TFRKPHN_pw_army_defence_factor = 0.05 }
		add_to_variable = { TFRKPHN_pw_war_support_factor = 0.03 }
		if = { limit = { NOT = { has_dynamic_modifier = { modifier = TFRKPHN_pw_legacy } } } add_dynamic_modifier = { modifier = TFRKPHN_pw_legacy } }
		force_update_dynamic_modifier = yes
	}
}

	custom_effect_tooltip = TFRKPHN_pw_grant_diplomacy_tt
	hidden_effect = {

	}
}
TFRKPHN_pw_migrate = {
	if = { limit = { has_idea = TFRKPHN_postwar_prosperity } remove_ideas = TFRKPHN_postwar_prosperity TFRKPHN_pw_grant_prosperity = yes }
	if = { limit = { has_idea = TFRKPHN_postwar_defense } remove_ideas = TFRKPHN_postwar_defense TFRKPHN_pw_grant_defense = yes }
	if = { limit = { has_idea = TFRKPHN_postwar_diplomacy } remove_ideas = TFRKPHN_postwar_diplomacy TFRKPHN_pw_grant_diplomacy = yes }
}
