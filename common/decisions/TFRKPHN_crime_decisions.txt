TFRKPHN_crime_category = {

	TFRKPHN_crime_seoul = {
		icon = GFX_decision_generic_political_actions
		cost = 35
		days_remove = 14
		visible = {
			check_variable = { TFRKPHN_crime_sel = 1 }
			NOT = { has_country_flag = TFRKPHN_crime_done_seoul }
		}
		available = {
			custom_trigger_tooltip = {
				tooltip = TFRKPHN_crime_left_tt
				check_variable = { TFRKPHN_crime_rate_seoul > 0 }
			}
		}
		complete_effect = {
			custom_effect_tooltip = TFRKPHN_crime_wait_tt
		}
		remove_effect = {
			custom_effect_tooltip = TFRKPHN_crime_op_tt
			hidden_effect = {
				set_temp_variable_to_random = { var = TFRKPHN_crime_drop min = 20 max = 31 integer = yes }
				clamp_temp_variable = { var = TFRKPHN_crime_drop min = 20 max = 30 }
				subtract_from_variable = { TFRKPHN_crime_rate_seoul = TFRKPHN_crime_drop }
				clamp_variable = { var = TFRKPHN_crime_rate_seoul min = 0 max = 100 }
				add_to_variable = { TFRKPHN_support_seoul = 1 }
				clamp_variable = { var = TFRKPHN_support_seoul min = 0 max = 100 }
				TFRKPHN_hong_update_support_display = yes
				TFRKPHN_crime_refresh = yes
				if = {
					limit = {
						check_variable = { TFRKPHN_crime_rate_seoul < 1 }
						NOT = { has_country_flag = TFRKPHN_crime_done_seoul }
					}
					set_country_flag = TFRKPHN_crime_done_seoul
					set_variable = { TFRKPHN_crime_last = 1 }
					add_to_variable = { TFRKPHN_crime_count = 1 }
					add_stability = 0.02
					add_to_variable = { TFRKPHN_support_seoul = 3 }
					clamp_variable = { var = TFRKPHN_support_seoul min = 0 max = 100 }
					TFRKPHN_hong_update_support_display = yes
					country_event = { id = tfrkphn.220 }
				}
			}
		}
		ai_will_do = { factor = 0 }
	}
