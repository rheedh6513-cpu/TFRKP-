			custom_trigger_tooltip = { tooltip = TFRKPHN_sim_one_at_a_time_tt NOT = { has_country_flag = TFRKPHN_sim_rf_running } }
		}
		complete_effect = {
			TFRKPHN_lij_tire_dec = yes
			custom_effect_tooltip = TFRKPHN_sim_rf_spend_tt

		}
		remove_effect = {
			custom_effect_tooltip = TFRKPHN_sim_rock_up_tt
			hidden_effect = { clr_country_flag = TFRKPHN_sim_rf_running add_to_variable = { TFRKPHN_sim_rock = 1 } clamp_variable = { var = TFRKPHN_sim_rock min = 0 max = 10 } set_variable = { TFRKPHN_sim_rock_f = TFRKPHN_sim_rock } add_to_variable = { TFRKPHN_sim_rock_f = 1 } }
		}
		ai_will_do = { factor = 0 }
