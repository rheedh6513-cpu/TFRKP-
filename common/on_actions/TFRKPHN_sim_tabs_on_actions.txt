# [안내] 심상정 정부 개혁 속도 검사(다달이)와 확보 의석(매일). sim_tabs_build.py가 만든다.
on_actions = {
	on_daily_KOR = {
		effect = {
			if = { limit = { original_tag = KOR has_country_flag = TFRKPHN_sim_reform_open } TFRKPHN_sim_seats_calc = yes }
		}
	}
	on_monthly_KOR = {
		effect = {
			if = {
				limit = { original_tag = KOR has_country_flag = TFRKPHN_sim_reform_open }
				if = { limit = { NOT = { has_variable = TFRKPHN_sim_rf_last } } set_variable = { TFRKPHN_sim_rf_last = 120 } }
				add_to_variable = { TFRKPHN_sim_rf_last = 30 }
				if = {
					limit = { check_variable = { TFRKPHN_sim_rf_last > 180 } NOT = { has_country_flag = TFRKPHN_sim_forum } check_variable = { TFRKPHN_sim_rock < 10 } }
					set_country_flag = TFRKPHN_sim_too_slow
					add_to_variable = { TFRKPHN_sim_people = -5 }
					clamp_variable = { var = TFRKPHN_sim_people min = 0 max = 100 }
				}
			}
		}
	}
}
