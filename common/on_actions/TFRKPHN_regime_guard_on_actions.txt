on_actions = {

		effect = {

			# 홍준표 정부: 연립 없이 보수주의 단독 집권
			if = { limit = { has_country_flag = TFRKP_HN_hong_president NOT = { has_country_flag = TFRKPHN_imp_done } has_government = conservative is_in_coalition = yes } end_coalition = yes }

				if = { limit = { NOT = { has_character = KOR_hong_joon_pyo_char } } recruit_character = KOR_hong_joon_pyo_char }
				KOR_hong_joon_pyo_char = { promote_character = korean_neoconservatism }
			}
			# 탄핵 뒤: 홍준표는 지도자 자리에 남지 않는다
			if = { limit = { has_country_flag = TFRKPHN_imp_done has_country_leader = { character = KOR_hong_joon_pyo_char ruling_only = yes } } KOR_hong_joon_pyo_char = { remove_country_leader_role = { ideology = korean_neoconservatism } } }

			if = { limit = { has_country_flag = TFRKPHN_imp_done NOT = { has_country_flag = TFRKPHN_samkim_end } }
				if = { limit = { has_country_flag = TFRKPHN_samkim_kms_pm } if = { limit = { NOT = { has_character = KOR_TFRKPHN_kms_char } } recruit_character = KOR_TFRKPHN_kms_char } KOR_TFRKPHN_kms_char = { promote_character = korean_neoconservatism } }
				else_if = { limit = { has_country_flag = TFRKPHN_samkim_kdy_pm } if = { limit = { NOT = { has_character = KOR_TFRKPHN_kdy_char } } recruit_character = KOR_TFRKPHN_kdy_char } KOR_TFRKPHN_kdy_char = { promote_character = centrist } }
				else_if = { limit = { has_country_flag = TFRKPHN_samkim_kbg_pm } if = { limit = { NOT = { has_character = KOR_TFRKPHN_kbg_char } } recruit_character = KOR_TFRKPHN_kbg_char } KOR_TFRKPHN_kbg_char = { promote_character = centrist } }

				else_if = { limit = { has_country_flag = TFRKPHN_e21_ahn has_character = KOR_ahn_cheol_soo_char } KOR_ahn_cheol_soo_char = { promote_character = classical_liberalism } }
				else_if = { limit = { has_country_flag = TFRKPHN_e21_sim_pres has_character = KOR_sim_sang_jung_char } KOR_sim_sang_jung_char = { promote_character = dreaming_realism } }
				else_if = { limit = { has_country_flag = TFRKPHN_kjt_fallen } if = { limit = { NOT = { has_character = KOR_TFRKPHN_woo_won_shik_char } } recruit_character = KOR_TFRKPHN_woo_won_shik_char } KOR_TFRKPHN_woo_won_shik_char = { promote_character = cooperative_democracy_liberal } }
				else_if = { limit = { has_country_flag = TFRKPHN_kmu_pres has_character = KOR_kim_moon_soo_char } KOR_kim_moon_soo_char = { promote_character = ultra_conservatism } }
				else_if = { limit = { has_country_flag = TFRKPHN_kjt_pres } if = { limit = { NOT = { has_character = KOR_TFRKPHN_kim_jin_tae_char } } recruit_character = KOR_TFRKPHN_kim_jin_tae_char } KOR_TFRKPHN_kim_jin_tae_char = { promote_character = ultra_conservatism } }
				else_if = { limit = { has_country_flag = TFRKPHN_e21_cho NOT = { has_country_flag = TFRKPHN_cho_removed } has_character = KOR_cho_won_jin_char } KOR_cho_won_jin_char = { promote_character = ultra_conservatism } }
				else_if = { limit = { always = yes has_character = KOR_lee_in_je_char } KOR_lee_in_je_char = { promote_character = korean_neoconservatism } }

			# 2027 대선 뒤 이인제, 오세훈 정부
			if = { limit = { has_country_flag = TFRKPHN_successor_lee_in_je NOT = { has_country_flag = TFRKPHN_lij_dead } has_government = conservative has_character = KOR_lee_in_je_char NOT = { has_country_leader = { character = KOR_lee_in_je_char ruling_only = yes } } } KOR_lee_in_je_char = { promote_character = korean_neoconservatism } }
			if = { limit = { has_country_flag = TFRKPHN_successor_oh_se_hoon has_government = conservative has_character = KOR_oh_se_hoon_char NOT = { has_country_leader = { character = KOR_oh_se_hoon_char ruling_only = yes } } } KOR_oh_se_hoon_char = { promote_character = korean_warm_conservatism } }

			}
		}
	}
