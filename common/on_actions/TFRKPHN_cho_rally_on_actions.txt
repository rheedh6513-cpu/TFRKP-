# [안내] 토요일 광장 탭: 조원진이 대통령이 되면 연다 (광장 50). 달마다 4씩 식는다. 조원진이 물러나면 국민정신을 걷는다. cho_build.py(cho_rally_part.py)가 만든다.
on_actions = {
	on_daily_KOR = {
		effect = {
			if = {
				limit = { has_country_flag = TFRKPHN_e21_cho NOT = { has_country_flag = TFRKPHN_cho_removed } NOT = { has_country_flag = TFRKPHN_kjt_pres } NOT = { has_country_flag = TFRKPHN_cr_init } }
				set_country_flag = TFRKPHN_cr_init
				set_variable = { TFRKPHN_cr = 50 }
				TFRKPHN_cr_refresh = yes
			}
			if = {
				limit = { has_country_flag = TFRKPHN_cr_init NOT = { has_country_flag = TFRKPHN_cr_closed } NOT = { has_country_flag = TFRKPHN_cr_tick } }
				set_country_flag = { flag = TFRKPHN_cr_tick days = 30 value = 1 }
				add_to_variable = { TFRKPHN_cr = -4 }
				TFRKPHN_cr_refresh = yes
			}
			if = {
				limit = { has_country_flag = TFRKPHN_cr_init NOT = { has_country_flag = TFRKPHN_cr_closed } OR = { has_country_flag = TFRKPHN_cho_removed has_country_flag = TFRKPHN_kjt_pres } }
				set_country_flag = TFRKPHN_cr_closed
				remove_dynamic_modifier = { modifier = TFRKPHN_cr_modifier }
			}
		}
	}
}
