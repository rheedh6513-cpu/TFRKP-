# [안내] 대아대전 시작(김정은 참수 뒤, regime_guard.py가 옮김)과 참전. 대아대전이 시작되면 한국은 반드시 PDTO 편에서 싸운다. 북한이 남아 있으면 북한이 쳐들어온다. gaw_sure.py가 만든다.
on_actions = {
	on_daily_KOR = {
		effect = {
			# 이미 대아대전이 시작된 저장 파일에서도 대아대전 중점 가지가 열리게 한다
			if = {
				limit = {
					has_global_flag = TFRKP_GAW_is_started_flag
					has_country_flag = TFRKPHN_kju_op_done
					NOT = { has_country_flag = KOR_great_asia_war }
				}
				set_country_flag = KOR_great_asia_war
				mark_focus_tree_layout_dirty = yes
			}
			# 김정은 참수 뒤 한국과 중국이 전쟁에 들어가면 대아대전이 시작된다 (Korea Plus의 대아대전 시작 처리와 같게)
			if = {
				limit = {
					has_country_flag = TFRKPHN_kju_op_done
					has_war_with = PRC
					NOT = { has_global_flag = TFRKP_GAW_is_started_flag }
					NOT = { AND = { has_global_flag = PRC_ongoing_taiwan_war NOT = { has_global_flag = PRC_taiwan_war_won } NOT = { has_global_flag = PRC_taiwan_war_lost } } }	# 양안전 중에는 대아대전이 아니다 (taiwan_fix.py)
				}
				set_global_flag = TFRKP_GAW_is_started_flag
				news_event = { id = tfrkphn.253 }
				if = { limit = { has_country_flag = TFRKPHN_successor_lee_in_je } country_event = { id = tfrkphn.567 days = 1 } }	# 새벽 세 시 (arc_build.py)
				# Korea Plus의 대아대전 중점 가지(공유 중점)가 홍준표, 이인제 트리에서도 열리게 한다
				set_country_flag = KOR_great_asia_war
				mark_focus_tree_layout_dirty = yes
				PRC = {
					unlock_national_focus = PRC_support_the_korean_way
					unlock_national_focus = PRC_finilize_korean_revolution
					remove_mission = PRC_korean_war_joining
					country_event = { id = china.1008 hours = 6 }
					if = {
						limit = { country_exists = JAP NOT = { has_war_with = JAP } }
						declare_war_on = { target = JAP type = annex_everything }
					}
					PRC_gaw_start = yes
				}
			}
			if = {
				limit = {
					original_tag = KOR
					NOT = { has_country_flag = TFRKPHN_gaw_in }
					NOT = { has_country_flag = TFRKPHN_ru_pdto_out }	# 러시아식 엔딩에서 PDTO를 나간 정권은 뺀다 (ru_end.py)
					OR = { has_global_flag = TFRKP_GAW_is_started_flag AND = { country_exists = JAP country_exists = PRC JAP = { has_war_with = PRC } NOT = { AND = { has_global_flag = PRC_ongoing_taiwan_war NOT = { has_global_flag = PRC_taiwan_war_won } NOT = { has_global_flag = PRC_taiwan_war_lost } } } } }	# 양안전 중 일중 전쟁은 대아대전이 아니다
					NOT = { has_global_flag = PRC_japanese_gaw_victory_global }
					NOT = { has_global_flag = PRC_chinese_gaw_victory_global }
				}
				set_country_flag = TFRKPHN_gaw_in
				if = {
					limit = { country_exists = JAP JAP = { is_faction_leader = yes } NOT = { is_in_faction_with = JAP } }
					if = { limit = { is_in_faction = yes } leave_faction = yes }
					JAP = { add_to_faction = KOR }
					KOR_add_to_pdto_effect = yes
				}
				if = {
					limit = { country_exists = PRK NOT = { has_war_with = PRK } }
					PRK = { declare_war_on = { target = KOR type = annex_everything } }
					country_event = { id = tfrkphn.7254 hours = 6 }
				}
				if = {
					limit = { country_exists = PRC NOT = { has_war_with = PRC } JAP = { has_war_with = PRC } NOT = { AND = { has_global_flag = PRC_ongoing_taiwan_war NOT = { has_global_flag = PRC_taiwan_war_won } NOT = { has_global_flag = PRC_taiwan_war_lost } } } }
					declare_war_on = { target = PRC type = annex_everything }
				}
			}
		}
	}
}
