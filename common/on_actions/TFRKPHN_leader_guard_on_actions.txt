# [안내] 지도자 지킴이와 홍준표 발전도 지우기. guard_build.py가 만든다.
on_actions = {
	on_daily_KOR = {
		effect = {
			if = { limit = { has_country_flag = TFRKPHN_imp_done has_dynamic_modifier = { modifier = TFRKPHN_dev_monthly } } remove_dynamic_modifier = { modifier = TFRKPHN_dev_monthly } }
			if = { limit = { has_country_flag = TFRKPHN_imp_done NOT = { has_country_flag = TFRKPHN_orp_formed } NOT = { has_country_flag = TFRKPHN_orp_7024_sched } } set_country_flag = TFRKPHN_orp_7024_sched country_event = { id = tfrkphn.7024 } }	# 탈당파가 아직 우리공화당으로 안 갔으면 (한 번만)
			if = { limit = { has_country_flag = TFRKPHN_orp_formed NOT = { has_country_flag = TFRKPHN_orp_fix24 } }
				set_country_flag = TFRKPHN_orp_fix24
				if = { limit = { check_variable = { KOR_NatAsm_ORP_seats > 24.5 } }
					set_temp_variable = { TFRKPHN_orp_x = KOR_NatAsm_ORP_seats }
					subtract_from_temp_variable = { TFRKPHN_orp_x = 24 }
					set_variable = { KOR_NatAsm_ORP_seats = 24 }
					add_to_variable = { KOR_NatAsm_PPP_seats = TFRKPHN_orp_x }
					KOR_rerender_NatAsm = yes
				}
			}
			if = { limit = { has_country_flag = TFRKPHN_imp_done }
				if = { limit = { has_character = KOR_lee_jae_myung_char } retire_character = KOR_lee_jae_myung_char }
				if = { limit = { has_character = KOR_moon_jae_in_char } retire_character = KOR_moon_jae_in_char }
			}
		}
	}
}
