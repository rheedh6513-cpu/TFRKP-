# 개혁 카드를 다 통과시켰을 때: 유승민의 독백 (게임 속 이야기. 2017년 공약, 득표율 6.8%, KDI 13년은 사실). 타협한 수에 따라 끝이 다르다
country_event = {
	id = tfrkphn.467
	title = tfrkphn.467.t
	desc = { text = tfrkphn.467.d0 trigger = { check_variable = { TFRKPHN_chb_nc < 0.5 } } }
	desc = { text = tfrkphn.467.d1 trigger = { check_variable = { TFRKPHN_chb_nc > 0.5 } check_variable = { TFRKPHN_chb_nc < 3.5 } } }
	desc = { text = tfrkphn.467.d2 trigger = { check_variable = { TFRKPHN_chb_nc > 3.5 } } }
	picture = GFX_TFRKPHN_lee_temp
	is_triggered_only = yes
	fire_only_once = yes
	option = {
		name = tfrkphn.467.a
		hidden_effect = { remove_ideas = KOR_dominance_of_chaebols remove_ideas = KOR_dominance_of_chaebols_2 }
		custom_effect_tooltip = tfrkphn.467.idea_tt
		if = { limit = { check_variable = { TFRKPHN_chb_nc > 3.5 } } add_ideas = TFRKPHN_chb_half }
		else_if = { limit = { check_variable = { TFRKPHN_chb_nc < 0.5 } } add_ideas = TFRKPHN_chb_sme }
		else_if = { limit = { NOT = { has_idea = KOR_copreation_of_chaebols } } add_ideas = KOR_copreation_of_chaebols }
		add_stability = 0.05
		custom_effect_tooltip = tfrkphn.467.tt
		hidden_effect = { add_to_variable = { TFRKPHN_job_n = TFRKPHN_chb_reward } TFRKPHN_job_update = yes }
		hidden_effect = { if = { limit = { check_variable = { TFRKPHN_chb_nc < 0.5 } } country_event = { id = tfrkphn.470 days = 2 } } country_event = { id = tfrkphn.495 days = 10 } }
	}
}
