	is_triggered_only = yes
	fire_only_once = yes
	option = {
		name = tfrkphn.472.a
		custom_effect_tooltip = tfrkphn.472.tt

			random_country = {
				limit = { has_country_flag = TFRKPHN_free_joseon exists = yes }
				every_owned_state = { KOR = { add_state_core = PREV } }
				KOR = { annex_country = { target = PREV transfer_troops = yes } }
			}
			every_owned_state = { limit = { OR = { is_core_of = PRK is_claimed_by = PRK } NOT = { is_core_of = KOR } } add_core_of = KOR }
			country_event = { id = tfrkphn.490 days = 3 }
		}
	}
}
