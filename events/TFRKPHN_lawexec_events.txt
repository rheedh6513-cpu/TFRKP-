# [안내] 이인제 정부: 중점 「온정주의 없는 법 집행」 이벤트 두 편. 문구는 localisation/korean/TFRKPHN_lawexec_l_korean.yml. lawexec_build.py가 만든다.
add_namespace = tfrkphn

# 전임 정부가 남긴 사형 집행 명령서 (홍준표의 찬성)
country_event = {
	id = tfrkphn.518
	title = tfrkphn.518.t
	desc = tfrkphn.518.d
	picture = GFX_TFRKPHN_lee_temp
	is_triggered_only = yes
	fire_only_once = yes
	option = {
		name = tfrkphn.518.a
		hidden_effect = { country_event = { id = tfrkphn.519 days = 1 } }
	}
}

# 다음 날: 대통령의 생각과 결정 (이인제의 반대)
country_event = {
	id = tfrkphn.519
	title = tfrkphn.519.t
	desc = tfrkphn.519.d
	picture = GFX_TFRKPHN_lee_temp
	is_triggered_only = yes
	fire_only_once = yes
	option = {	# 돌려보내고 종신형 입법
		name = tfrkphn.519.a
		add_political_power = -15
		add_stability = 0.01
		hidden_effect = { country_event = { id = tfrkphn.520 days = 1 } }
	}
	option = {	# 금고에 넣어 둔다
		name = tfrkphn.519.b
		add_political_power = 10
		hidden_effect = { country_event = { id = tfrkphn.520 days = 1 } }
	}
	option = {	# 서명한다: 형법이 「처형」(medium_prison, Capital Punishment)으로 바뀐다. TFR 본체 글 파일에서 이름을 확인했다
		name = tfrkphn.519.c
		add_ideas = medium_prison
		add_political_power = 20
		add_stability = -0.02
		set_country_flag = TFRKPHN_exec_signed
		hidden_effect = { country_event = { id = tfrkphn.520 days = 1 } }
	}
}

# 그다음 날: 홍준표 전 대통령의 글. 서명했는지에 따라 글이 다르다
country_event = {
	id = tfrkphn.520
	title = tfrkphn.520.t
	desc = { text = tfrkphn.520.d_yes trigger = { has_country_flag = TFRKPHN_exec_signed } }
	desc = { text = tfrkphn.520.d_no trigger = { NOT = { has_country_flag = TFRKPHN_exec_signed } } }
	picture = GFX_TFRKPHN_lee_temp
	is_triggered_only = yes
	fire_only_once = yes
	option = {
		name = tfrkphn.520.a
	}
}
