# 2021 People Power Party convention: Yoon Suk-yeol or Hong Joon-pyo
country_event = {
	id = tfrkphn.1
	title = tfrkphn.1.t
	desc = tfrkphn.1.desc
	picture = GFX_TFRKPHN_ppp_convention

	fire_only_once = yes

	trigger = {
		tag = KOR
		NOT = { date < 2021.11.5 }
		# 이미 대선 후보를 골랐다면(대선 중점을 일찍 끝낸 경우) 늦은 전당대회는 열리지 않는다
		NOT = { has_country_flag = KOR_sided_with_yoon }
		NOT = { has_country_flag = KOR_sided_with_lee }
	}

	mean_time_to_happen = { days = 1 }

	immediate = {
		set_global_flag = TFRKPHN_ppp_convention_fired
	}

	option = {
		name = tfrkphn.1.a
		# AI: 이재명이면 윤석열, 이낙연이면 홍준표. 상대 당이 아직이면 반반
		ai_chance = {
			base = 50
			modifier = { factor = 100 has_country_flag = TFRKP_HN_DPK_lee_jaemyung }
			modifier = { factor = 0 has_country_flag = TFRKP_HN_DPK_lee_nakyeon }
		}
		custom_effect_tooltip = TFRKPHN_tt_yoon
		set_country_flag = TFRKP_HN_PPP_yoon
	}

	option = {
		name = tfrkphn.1.b
		ai_chance = {
			base = 50
			modifier = { factor = 100 has_country_flag = TFRKP_HN_DPK_lee_nakyeon }
			modifier = { factor = 0 has_country_flag = TFRKP_HN_DPK_lee_jaemyung }
		}

	}

# 홍준표 시작 처리 안전장치 (매일 점검 on_action이 호출한다)
# - 홍준표 대통령 표시가 없으면 시작 처리를 직접 실행한다 (tfrkphn.10 이 돌지 않았을 때)
# - 표시는 있는데 홍준표 트리가 적용되지 않았으면 트리를 다시 불러온다
country_event = {
	id = tfrkphn.12
	hidden = yes
	is_triggered_only = yes

	immediate = {
		log = "TFRKPHN_DIAG tfrkphn.12 safety net fired"
		add_to_variable = { TFRKPHN_tree_retry = 1 }
		if = {
			limit = { NOT = { has_country_flag = TFRKP_HN_hong_president } }
			log = "TFRKPHN_DIAG tfrkphn.12: running hong start effect directly"
			TFRKPHN_hong_start_effect = yes
		}
		else = {
			log = "TFRKPHN_DIAG tfrkphn.12: reloading hong tree"
			load_focus_tree = { tree = TFRKPHN_hong_tree keep_completed = no }
			mark_focus_tree_layout_dirty = yes
		}
	}

	option = {
		name = tfrkphn.12.a
	}
}
