# 홍준표 시작 처리 안전장치 (매일 점검 on_action이 호출한다)
# - 홍준표 대통령 표시가 없으면 시작 처리를 직접 실행한다 (tfrkphn.10 이 돌지 않았을 때)
# - 표시는 있는데 홍준표 트리가 적용되지 않았으면 트리를 다시 불러온다
country_event = {
	id = tfrkphn.12
	hidden = yes
	is_triggered_only = yes

	immediate = {
		log = "TFRKPHN_DIAG tfrkphn.12 safety net fired"
		add_to_variable = { TFRKPHN_tree_retry = 1 }
		if = {
			limit = { NOT = { has_country_flag = TFRKP_HN_hong_president } }
			log = "TFRKPHN_DIAG tfrkphn.12: running hong start effect directly"
			TFRKPHN_hong_start_effect = yes
		}
		else = {
			log = "TFRKPHN_DIAG tfrkphn.12: reloading hong tree"
			load_focus_tree = { tree = TFRKPHN_hong_tree keep_completed = no }
			mark_focus_tree_layout_dirty = yes
		}
	}

	option = {
		name = tfrkphn.12.a
	}
}
