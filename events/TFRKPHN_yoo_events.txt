# 통일세 발표 나흘 뒤: 세금 이야기. 고른 안에 따라 글이 다르다
country_event = {
	id = tfrkphn.512
	title = tfrkphn.512.t
	desc = { text = tfrkphn.512.d_full trigger = { has_country_flag = TFRKPHN_unitax_full } }
	desc = { text = tfrkphn.512.d_half trigger = { has_country_flag = TFRKPHN_unitax_half } }
	desc = { text = tfrkphn.512.d_none trigger = { NOT = { has_country_flag = TFRKPHN_unitax_full } NOT = { has_country_flag = TFRKPHN_unitax_half } } }
	picture = GFX_TFRKPHN_lee_temp
	is_triggered_only = yes
	fire_only_once = yes
	option = {
		name = tfrkphn.512.a
		trigger = { OR = { has_country_flag = TFRKPHN_unitax_full has_country_flag = TFRKPHN_unitax_half } }
		add_stability = 0.01
	}
	option = {
		name = tfrkphn.512.b
		trigger = { NOT = { has_country_flag = TFRKPHN_unitax_full } NOT = { has_country_flag = TFRKPHN_unitax_half } }
		add_political_power = -10
	}
}
